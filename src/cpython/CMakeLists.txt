# yetty_cpython — a libpython-free SUBSET of the CPython 3.14 parser.
#
# Compiles the vendored CPython PEG parser + tokenizer against our own
# libpython-free runtime (src/runtime) and a plain-C AST generated from the
# ASDL schema (src/ast). Produces an AST + ast.dump()-style output, linking
# against libc only — no libpython. See README.md for scope and limitations.
#
# The runtime headers under src/runtime/ SHADOW CPython's <Python.h> and
# pycore_*.h, so they are kept strictly PRIVATE to these targets and never
# exposed to the rest of the yetty build.

set(CPYTHON_DIR ${CMAKE_CURRENT_SOURCE_DIR})
set(CPYTHON_VEN ${CPYTHON_DIR}/third_party/cpython/Parser)

add_library(yetty_cpython STATIC
    # libpython-free runtime
    ${CPYTHON_DIR}/src/runtime/object.c
    ${CPYTHON_DIR}/src/runtime/unicode.c
    ${CPYTHON_DIR}/src/runtime/numbers.c
    ${CPYTHON_DIR}/src/runtime/bytes.c
    ${CPYTHON_DIR}/src/runtime/errors.c
    ${CPYTHON_DIR}/src/runtime/mem.c
    # plain-C AST + dumper (generated by tools/gen_ast.py + helpers)
    ${CPYTHON_DIR}/src/ast/arena.c
    ${CPYTHON_DIR}/src/ast/ast.gen.c
    ${CPYTHON_DIR}/src/ast/ast_dump.gen.c
    ${CPYTHON_DIR}/src/ast/pyp_dump.c
    ${CPYTHON_DIR}/src/ast/printable.gen.c
    # clean public-API wrapper
    ${CPYTHON_DIR}/src/cpython_api.c
    # vendored CPython parser + tokenizer (PSF License — see third_party/)
    ${CPYTHON_VEN}/token.c
    ${CPYTHON_VEN}/pegen.c
    ${CPYTHON_VEN}/pegen_errors.c
    ${CPYTHON_VEN}/action_helpers.c
    ${CPYTHON_VEN}/string_parser.c
    ${CPYTHON_VEN}/peg_api.c
    ${CPYTHON_VEN}/parser.c
    ${CPYTHON_VEN}/lexer/buffer.c
    ${CPYTHON_VEN}/lexer/lexer.c
    ${CPYTHON_VEN}/lexer/state.c
    ${CPYTHON_VEN}/tokenizer/string_tokenizer.c
    ${CPYTHON_VEN}/tokenizer/utf8_tokenizer.c
    ${CPYTHON_VEN}/tokenizer/file_tokenizer.c
    ${CPYTHON_VEN}/tokenizer/helpers.c
)

set_target_properties(yetty_cpython PROPERTIES C_STANDARD 11 C_EXTENSIONS ON)

# Internal include paths — PRIVATE so the shadow headers never leak.
target_include_directories(yetty_cpython PRIVATE
    ${CPYTHON_DIR}/src/runtime
    ${CPYTHON_DIR}/src/ast
    ${CPYTHON_VEN}
)
# Public: only the clean wrapper header (include/yetty/cpython/cpython.h).
target_include_directories(yetty_cpython PUBLIC ${YETTY_ROOT}/include)

target_compile_definitions(yetty_cpython PRIVATE Py_BUILD_CORE)
# Vendored third-party C: silence warnings, never treat them as errors here.
target_compile_options(yetty_cpython PRIVATE -w)

# Standalone tool: py-parse <file.py> -> AST dump on stdout.
add_executable(py-parse ${CPYTHON_DIR}/src/main_free.c)
set_target_properties(py-parse PROPERTIES C_STANDARD 11 C_EXTENSIONS ON)
target_include_directories(py-parse PRIVATE
    ${CPYTHON_DIR}/src/runtime ${CPYTHON_DIR}/src/ast ${CPYTHON_VEN})
target_compile_definitions(py-parse PRIVATE Py_BUILD_CORE)
target_compile_options(py-parse PRIVATE -w)
target_link_libraries(py-parse PRIVATE yetty_cpython)
