# yrdawn — WebGPU-over-OSC bridge.
#
# Two consumers:
#   * remote process (yetty-webasm builds) — links yetty_yrdawn, which
#     implements <webgpu/webgpu.h> by encoding each call as an OSC frame
#     on stdout. Pure C, no Dawn, no C++.
#   * local yetty (desktop/android builds) — yetty_yterm links yetty_yrdawn
#     for the protocol contract; the codegen-emitted server dispatcher
#     (server-dispatch.gen.c, kept in this directory) is added to
#     yetty_yterm's sources by src/yetty/yterm/CMakeLists.txt.
#
# Generated sources (methods.gen.h, *.gen.c) are committed to the repo
# and regenerated by hand via `python3 src/yetty/yrdawn/generate.py`
# whenever the METHODS table in generate.py changes. CMake treats them
# as regular sources — no build-time codegen, no Python dependency.

add_library(yetty_yrdawn STATIC
    wire.c
    client.c
    client-stubs.gen.c
)

# Emscripten's clang treats incompatible-pointer-types as a hard error;
# Dawn-targeted clangs only warn. The codegen produces `char *` for
# struct members typed `const char *const *` (e.g.
# WGPUDawnTogglesDescriptor::enabledToggles) — same code, same Dawn
# headers, only emcc rejects it. Downgrade on webasm to match the
# behaviour of every other host until the codegen is fixed.
if(EMSCRIPTEN)
    target_compile_options(yetty_yrdawn PRIVATE
        -Wno-error=incompatible-pointer-types
        -Wno-incompatible-pointer-types
    )
endif()

target_include_directories(yetty_yrdawn BEFORE PUBLIC
    # OUR committed webgpu.h MUST be searched first — it's the canonical
    # type surface the bridge protocol is built against. Every Dawn
    # prebuilt ships a slightly different webgpu.h (Android omits Drm /
    # Dawn color-space extensions, webasm omits Vulkan handles, etc.).
    # If client and server resolved `<webgpu/webgpu.h>` to whatever Dawn
    # ships on each platform, struct layouts would diverge and the wire
    # format would silently corrupt. Pinning the header here guarantees:
    # (a) the committed .gen.c always compiles, (b) the wire layout is
    # identical on every host that links yetty_yrdawn. Regenerate
    # (`python3 src/yetty/yrdawn/generate.py`) against this same header.
    # BEFORE is load-bearing: webgpu's INTERFACE include dir comes in
    # via yetty_yterm's link line, and without BEFORE the dedup could
    # pick Dawn's webgpu.h ahead of ours on some generator backends.
    ${YETTY_ROOT}/include/yetty/yrdawn
)
target_include_directories(yetty_yrdawn PUBLIC
    ${YETTY_ROOT}/include
)

target_link_libraries(yetty_yrdawn
    PUBLIC
        yetty_api_yrdawn
        yetty_ycore
        yetty_yface
        # Figure-tree emit path: the client connects to the host, navigates
        # to the root container (yetty_api_yterminal), and drives
        # create/delete/body through the typed container stubs. It is a pure
        # remote-driver — only the api stubs, never the yfigure impl.
        yetty_api_yfigure
        yetty_api_yterminal
        # Cross-platform sleep + non-blocking read used by the client
        # pump. yetty_yplatform_core is per-OS so we don't drag POSIX
        # (poll(2), nanosleep) into the Windows build.
        yetty_yplatform_core
)

# yetty_yrdawn_server — desktop-side host of remote canvases.
#
# Holds the yfigure subclass (figure.c) and the per-client shared state
# (session.c). The codegen dispatcher (server-dispatch.gen.c, or the
# UNKNOWN_METHOD stub on platforms whose Dawn prebuilt is incomplete)
# lives next to it. Hosts (yetty_yterm) link this library to get the
# register_factory entrypoint; yframework's register_figure_factories
# resolves it at link time.
#
# Skipped on webasm: the client end of the bridge runs in the wasm
# process; the server only ever lives on the host that's hosting
# canvases. The .gen.c also has incompatible-pointer-types issues
# under emscripten's clang.
if(NOT EMSCRIPTEN)
    set(_yrdawn_server_dispatch ${CMAKE_CURRENT_SOURCE_DIR}/server-dispatch.gen.c)
    # Same reasoning as the old yterm/CMakeLists.txt logic: mobile Dawn
    # prebuilts don't export every entrypoint the generated dispatcher
    # calls. Swap in the stub on those targets so the link succeeds and
    # CMDs simply answer UNKNOWN_METHOD.
    if(ANDROID OR YETTY_ANDROID OR YETTY_IOS OR YETTY_TVOS)
        set(_yrdawn_server_dispatch ${CMAKE_CURRENT_SOURCE_DIR}/server-dispatch-stub.c)
    endif()

    add_library(yetty_yrdawn_server STATIC
        figure.c
        session.c
            ${_yrdawn_server_dispatch}
    )

    target_include_directories(yetty_yrdawn_server BEFORE PUBLIC
        ${YETTY_ROOT}/include/yetty/yrdawn
    )
    target_include_directories(yetty_yrdawn_server PUBLIC
        ${YETTY_ROOT}/include
    )
    # Internal yclass codegen headers (methods.gen.h) sit under src/.
    target_include_directories(yetty_yrdawn_server PRIVATE ${YETTY_ROOT}/src)

    target_link_libraries(yetty_yrdawn_server PUBLIC
        yetty_ycore
        yetty_yclass
        yetty_yfigure
        yetty_yframework
        yetty_yrender
        yetty_ywire
        webgpu
    )

    # figure.c carries `[[clang::annotate(...)]]` on its data struct
    # (codegen reads it off the AST). The C2x attribute syntax needs C23
    # to parse — host clang/gcc accept it by default, NDK clang 17 does
    # not. Off-MSVC only: MSVC can't express C23 via CMake's C_STANDARD
    # and parses the attributes via the global /std:clatest. (Mirrors
    # src/yetty/ygui/CMakeLists.txt.)
    if(NOT MSVC)
        set_target_properties(yetty_yrdawn_server PROPERTIES
            C_STANDARD 23
            C_STANDARD_REQUIRED ON
        )
    endif()
endif()

# Object-API library (role-split codegen layout): the typed stubs alone,
# with no strong references to the implementation, accessors, skels or
# registration. Pure wire consumers link THIS; the impl lib links it too
# (skels re-enter the public stubs). Header: <yetty/api/yrdawn/*.h>.
add_library(yetty_api_yrdawn STATIC
    ${YETTY_ROOT}/src/yetty/gen/api/yrdawn/figure.c
)

target_include_directories(yetty_api_yrdawn PUBLIC
    ${YETTY_ROOT}/include
)

target_link_libraries(yetty_api_yrdawn PUBLIC
    yetty_ycore
    yetty_yclass
)

# Register with the yclass codegen graph — the module list IS these
# call sites (see build-tools/yetty/yclass-codegen.cmake).
yetty_yclass_module(yrdawn)
