# yrdawn — WebGPU-over-OSC bridge.
#
# Two consumers:
#   * remote process (yetty-webasm builds) — links yetty_yrdawn, which
#     implements <webgpu/webgpu.h> by encoding each call as an OSC frame
#     on stdout. Pure C, no Dawn, no C++.
#   * local yetty (desktop/android builds) — yetty_yterm links yetty_yrdawn
#     for the protocol contract; the codegen-emitted server dispatcher
#     (server-dispatch.gen.c, kept in this directory) is added to
#     yetty_yterm's sources by src/yetty/yterm/CMakeLists.txt.
#
# Generated sources (methods.gen.h, *.gen.c) are committed to the repo
# and regenerated by hand via `python3 src/yetty/yrdawn/generate.py`
# whenever the METHODS table in generate.py changes. CMake treats them
# as regular sources — no build-time codegen, no Python dependency.

add_library(yetty_yrdawn STATIC
    wire.c
    client.c
    client-stubs.gen.c
)

target_include_directories(yetty_yrdawn PUBLIC
    ${YETTY_ROOT}/include
    # Header-only consumption of webgpu.h: the codegen-emitted args
    # structs reference WGPU* POD types by value (WGPUColor,
    # WGPUExtent3D, …). We pull in the include path from the `webgpu`
    # CMake target without linking the Dawn library — the client side
    # stays Dawn-free; the server side links Dawn via yetty_yterm.
    $<TARGET_PROPERTY:webgpu,INTERFACE_INCLUDE_DIRECTORIES>
)

target_link_libraries(yetty_yrdawn
    PUBLIC
        yetty_ycore
        yetty_yface
)
